import Mathlib

namespace GoldbachRound17.FourFormRoots
open Finset
noncomputable section

def actualForm (l N e p0 : ℕ) (x : ZMod l) : ZMod l :=
  x * ((N : ZMod l) - (e : ZMod l) * x) * ((N : ZMod l) - x) *
    ((N : ZMod l) - (p0 : ZMod l) * x)

def residueUniverse (l : ℕ) : Finset (ZMod l) :=
  (Finset.range l).image fun r : ℕ => (r : ZMod l)

def actualRoots (l N e p0 : ℕ) : Finset (ZMod l) :=
  (residueUniverse l).filter fun x => actualForm l N e p0 x = 0

def actualRho (l N e p0 : ℕ) : ℕ := (actualRoots l N e p0).card

def localDensity (l N e p0 : ℕ) : ℝ := (actualRho l N e p0 : ℝ) / l

def candidates (l N e p0 : ℕ) [Fact l.Prime] : Finset (ZMod l) :=
  {0, (N : ZMod l), (N : ZMod l) / (e : ZMod l),
    (N : ZMod l) / (p0 : ZMod l)}

def collisionProduct (l N e p0 : ℕ) : ZMod l :=
  (N : ZMod l) * (e : ZMod l) * (p0 : ZMod l) * ((e : ZMod l) - 1) *
    ((e : ZMod l) - (p0 : ZMod l)) * ((p0 : ZMod l) - 1)

def natDelta (N e p0 : ℕ) : ℕ := N * e * p0 * (e - 1) * (e - p0) * (p0 - 1)

def integerForm (N e p0 q : ℕ) : ℤ :=
  (q : ℤ) * ((N : ℤ) - (e : ℤ) * q) * ((N : ℤ) - q) *
    ((N : ℤ) - (p0 : ℤ) * q)

def fourFormRoughCell (J P : Finset ℕ) (N e p0 : ℕ) : Finset ℕ :=
  J.filter fun q => ∀ l ∈ P, ¬ (l : ℤ) ∣ integerForm N e p0 q

variable {l N e p0 : ℕ} [Fact l.Prime]

theorem residueUniverse_eq_univ : residueUniverse l = Finset.univ := by
  ext x
  simp only [residueUniverse, Finset.mem_image, Finset.mem_range, Finset.mem_univ,
    iff_true]
  exact ⟨x.val, ZMod.val_lt x, ZMod.natCast_zmod_val x⟩

theorem mem_actualRoots (x : ZMod l) :
    x ∈ actualRoots l N e p0 ↔ actualForm l N e p0 x = 0 := by
  simp [actualRoots, residueUniverse_eq_univ]

theorem zero_mem_actualRoots : (0 : ZMod l) ∈ actualRoots l N e p0 := by
  simp [actualRoots, actualForm, residueUniverse_eq_univ]

theorem rho_lower : 1 ≤ actualRho l N e p0 := by
  exact Finset.one_le_card.mpr ⟨0, zero_mem_actualRoots⟩

theorem rho_le_modulus : actualRho l N e p0 ≤ l := by
  have h := Finset.card_le_card (Finset.subset_univ (actualRoots l N e p0))
  simpa [actualRho, ZMod.card] using h

theorem coprime_cast_not_both_zero (h : e.Coprime N) :
    ¬ ((e : ZMod l) = 0 ∧ (N : ZMod l) = 0) := by
  rintro ⟨he, hN⟩
  have dle := (ZMod.natCast_zmod_eq_zero_iff_dvd e l).mp he
  have dlN := (ZMod.natCast_zmod_eq_zero_iff_dvd N l).mp hN
  exact (Nat.not_coprime_of_dvd_of_dvd (Fact.out : l.Prime).one_lt dle dlN) h

theorem linear_root_eq_div_of_nonzero {x : ZMod l} (he : (e : ZMod l) ≠ 0)
    (hx : (N : ZMod l) - (e : ZMod l) * x = 0) :
    x = (N : ZMod l) / (e : ZMod l) := by
  apply (eq_div_iff he).mpr
  simpa [mul_comm] using (sub_eq_zero.mp hx).symm

theorem linear_root_eq_div {x : ZMod l} (h : e.Coprime N)
    (hx : (N : ZMod l) - (e : ZMod l) * x = 0) :
    x = (N : ZMod l) / (e : ZMod l) := by
  have he : (e : ZMod l) ≠ 0 := by
    intro hz
    have hN : (N : ZMod l) = 0 := by simpa [hz] using hx
    exact coprime_cast_not_both_zero h ⟨hz, hN⟩
  exact linear_root_eq_div_of_nonzero he hx

theorem roots_subset_candidates (heN : e.Coprime N) (hpN : p0.Coprime N) :
    actualRoots l N e p0 ⊆ candidates l N e p0 := by
  intro x hx
  have hf := (mem_actualRoots x).mp hx
  simp only [actualForm, mul_eq_zero] at hf
  rcases hf with ((hx0 | hxe) | hxN) | hxp
  · simp [candidates, hx0]
  · simp [candidates, linear_root_eq_div heN hxe]
  · simp [candidates, sub_eq_zero.mp hxN]
  · simp [candidates, linear_root_eq_div hpN hxp]

theorem candidates_card_le_four : (candidates l N e p0).card ≤ 4 := by
  have h1 := Finset.card_insert_le (0 : ZMod l)
    ({(N : ZMod l), (N : ZMod l) / (e : ZMod l), (N : ZMod l) / (p0 : ZMod l)} : Finset (ZMod l))
  have h2 := Finset.card_insert_le (N : ZMod l)
    ({(N : ZMod l) / (e : ZMod l), (N : ZMod l) / (p0 : ZMod l)} : Finset (ZMod l))
  have h3 := Finset.card_insert_le ((N : ZMod l) / (e : ZMod l))
    ({(N : ZMod l) / (p0 : ZMod l)} : Finset (ZMod l))
  simp only [Finset.card_singleton] at h3
  dsimp [candidates]
  omega

theorem rho_le_four (heN : e.Coprime N) (hpN : p0.Coprime N) :
    actualRho l N e p0 ≤ 4 := by
  exact (Finset.card_le_card (roots_subset_candidates heN hpN)).trans candidates_card_le_four

theorem rho_bounds (heN : e.Coprime N) (hpN : p0.Coprime N) :
    1 ≤ actualRho l N e p0 ∧ actualRho l N e p0 ≤ min 4 l := by
  exact ⟨rho_lower, le_min (rho_le_four heN hpN) rho_le_modulus⟩

theorem roots_eq_candidates_of_nonzero (he : (e : ZMod l) ≠ 0)
    (hp : (p0 : ZMod l) ≠ 0) :
    actualRoots l N e p0 = candidates l N e p0 := by
  ext x
  simp only [mem_actualRoots, candidates, Finset.mem_insert, Finset.mem_singleton]
  constructor
  · intro hf
    simp only [actualForm, mul_eq_zero] at hf
    rcases hf with ((hx0 | hxe) | hxN) | hxp
    · exact Or.inl hx0
    · exact Or.inr (Or.inr (Or.inl (linear_root_eq_div_of_nonzero he hxe)))
    · exact Or.inr (Or.inl (sub_eq_zero.mp hxN).symm)
    · exact Or.inr (Or.inr (Or.inr (linear_root_eq_div_of_nonzero hp hxp)))
  · rintro (hx0 | hxN | hxe | hxp)
    · simp [actualForm, hx0]
    · simp [actualForm, hxN]
    · rw [hxe]
      have hmul : (e : ZMod l) * ((N : ZMod l) / (e : ZMod l)) = N := by
        field_simp
      simp [actualForm, hmul]
    · rw [hxp]
      have hmul : (p0 : ZMod l) * ((N : ZMod l) / (p0 : ZMod l)) = N := by
        field_simp
      simp [actualForm, hmul]

theorem density_positive : 0 < localDensity l N e p0 := by
  have hl : 0 < (l : ℝ) := by exact_mod_cast (Fact.out : l.Prime).pos
  have hr : 0 < (actualRho l N e p0 : ℝ) := by exact_mod_cast rho_lower
  exact div_pos hr hl

theorem density_lt_one_of_nonsaturated (h : actualRho l N e p0 < l) :
    localDensity l N e p0 < 1 := by
  have hl : 0 < (l : ℝ) := by exact_mod_cast (Fact.out : l.Prime).pos
  apply (div_lt_one hl).mpr
  exact_mod_cast h

theorem rho_eq_four_of_collision_nonzero (h : collisionProduct l N e p0 ≠ 0) :
    actualRho l N e p0 = 4 := by
  simp only [collisionProduct, mul_ne_zero_iff] at h
  rcases h with ⟨⟨⟨⟨⟨hN, he⟩, hp⟩, he1⟩, hep⟩, hp1⟩
  have hNe : (N : ZMod l) ≠ (N : ZMod l) / (e : ZMod l) := by
    intro hEq
    have hm := (eq_div_iff he).mp hEq
    have heq : (e : ZMod l) = 1 :=
      mul_left_cancel₀ hN (by simpa using hm)
    exact he1 (sub_eq_zero.mpr heq)
  have hNp : (N : ZMod l) ≠ (N : ZMod l) / (p0 : ZMod l) := by
    intro hEq
    have hm := (eq_div_iff hp).mp hEq
    have hpq : (p0 : ZMod l) = 1 :=
      mul_left_cancel₀ hN (by simpa using hm)
    exact hp1 (sub_eq_zero.mpr hpq)
  have hdiv : (N : ZMod l) / (e : ZMod l) ≠ (N : ZMod l) / (p0 : ZMod l) := by
    intro hEq
    have hpe : (p0 : ZMod l) = (e : ZMod l) :=
      mul_left_cancel₀ hN ((div_eq_div_iff he hp).mp hEq)
    exact hep (sub_eq_zero.mpr hpe.symm)
  have hz : (0 : ZMod l) ∉
      ({(N : ZMod l), (N : ZMod l) / (e : ZMod l),
        (N : ZMod l) / (p0 : ZMod l)} : Finset (ZMod l)) := by
    simp [Ne.symm hN, Ne.symm (div_ne_zero hN he), Ne.symm (div_ne_zero hN hp)]
  have hNc : (N : ZMod l) ∉
      ({(N : ZMod l) / (e : ZMod l), (N : ZMod l) / (p0 : ZMod l)} : Finset (ZMod l)) := by
    simp [hNe, hNp]
  have hepc : (N : ZMod l) / (e : ZMod l) ∉
      ({(N : ZMod l) / (p0 : ZMod l)} : Finset (ZMod l)) := by simp [hdiv]
  unfold actualRho
  rw [roots_eq_candidates_of_nonzero he hp, candidates,
    Finset.card_insert_of_not_mem hz, Finset.card_insert_of_not_mem hNc,
    Finset.card_insert_of_not_mem hepc, Finset.card_singleton]

theorem natDelta_cast (h1 : 1 ≤ p0) (hpe : p0 ≤ e) :
    (natDelta N e p0 : ZMod l) = collisionProduct l N e p0 := by
  simp only [natDelta, Nat.cast_mul, Nat.cast_sub (h1.trans hpe), Nat.cast_sub hpe,
    Nat.cast_sub h1, Nat.cast_one, collisionProduct]

theorem rho_eq_four_of_not_dvd_delta (h1 : 1 ≤ p0) (hpe : p0 ≤ e)
    (h : ¬ l ∣ natDelta N e p0) : actualRho l N e p0 = 4 := by
  have hc : (natDelta N e p0 : ZMod l) ≠ 0 := by
    intro hz
    exact h ((ZMod.natCast_zmod_eq_zero_iff_dvd (natDelta N e p0) l).mp hz)
  rw [natDelta_cast h1 hpe] at hc
  exact rho_eq_four_of_collision_nonzero hc

theorem saturated_roots_eq_univ (h : actualRho l N e p0 = l) :
    actualRoots l N e p0 = Finset.univ := by
  apply Finset.eq_of_subset_of_card_le (Finset.subset_univ _)
  simpa [actualRho, ZMod.card] using h.ge

theorem saturation_forces_actualForm_zero (h : actualRho l N e p0 = l)
    (x : ZMod l) : actualForm l N e p0 x = 0 := by
  apply (mem_actualRoots x).mp
  rw [saturated_roots_eq_univ h]
  exact Finset.mem_univ x

theorem nonsaturated_of_actualForm_nonzero {x : ZMod l}
    (h : actualForm l N e p0 x ≠ 0) : actualRho l N e p0 < l := by
  apply lt_of_le_of_ne rho_le_modulus
  intro hEq
  exact h (saturation_forces_actualForm_zero hEq x)

theorem roots_of_N_zero (heN : e.Coprime N) (hpN : p0.Coprime N)
    (hN : (N : ZMod l) = 0) : actualRoots l N e p0 = {0} := by
  have he : (e : ZMod l) ≠ 0 := fun hz => coprime_cast_not_both_zero heN ⟨hz, hN⟩
  have hp : (p0 : ZMod l) ≠ 0 := fun hz => coprime_cast_not_both_zero hpN ⟨hz, hN⟩
  rw [roots_eq_candidates_of_nonzero he hp]
  simp [candidates, hN]

theorem rho_eq_one_of_dvd_N (heN : e.Coprime N) (hpN : p0.Coprime N)
    (hN : l ∣ N) : actualRho l N e p0 = 1 := by
  unfold actualRho
  rw [roots_of_N_zero heN hpN ((ZMod.natCast_zmod_eq_zero_iff_dvd N l).mpr hN)]
  exact Finset.card_singleton 0

instance primeThreeFact : Fact (Nat.Prime 3) := ⟨by decide⟩

theorem roots_mod_three_e_two (hN : (N : ZMod 3) = 1)
    (he : (e : ZMod 3) = 2) : actualRoots 3 N e 3 = Finset.univ := by
  ext x
  simp only [mem_actualRoots, Finset.mem_univ, iff_true]
  have h3 : (3 : ZMod 3) = 0 := ZMod.natCast_self 3
  have h2 : (2 : ZMod 3) = -1 := by linear_combination h3
  simp only [actualForm, Nat.cast_ofNat, hN, he, h2, h3, zero_mul, sub_zero, mul_one]
  have hpow : x ^ 3 = x := ZMod.pow_card x
  linear_combination -hpow

theorem rho_mod_three_e_two (hN : (N : ZMod 3) = 1)
    (he : (e : ZMod 3) = 2) : actualRho 3 N e 3 = 3 := by
  unfold actualRho
  rw [roots_mod_three_e_two hN he]
  simp [ZMod.card]

theorem roots_mod_three_e_zero (hN : (N : ZMod 3) = 1)
    (he : (e : ZMod 3) = 0) : actualRoots 3 N e 3 = {0,1} := by
  ext x
  have h3 : (3 : ZMod 3) = 0 := ZMod.natCast_self 3
  simp [mem_actualRoots, actualForm, hN, he, h3, mul_eq_zero, sub_eq_zero, zero_eq_sub, eq_comm]

theorem roots_mod_three_e_one (hN : (N : ZMod 3) = 1)
    (he : (e : ZMod 3) = 1) : actualRoots 3 N e 3 = {0,1} := by
  ext x
  have h3 : (3 : ZMod 3) = 0 := ZMod.natCast_self 3
  simp [mem_actualRoots, actualForm, hN, he, h3, mul_eq_zero, sub_eq_zero, zero_eq_sub, eq_comm]

theorem rho_mod_three_e_zero (hN : (N : ZMod 3) = 1)
    (he : (e : ZMod 3) = 0) : actualRho 3 N e 3 = 2 := by
  unfold actualRho
  rw [roots_mod_three_e_zero hN he]
  norm_num

theorem rho_mod_three_e_one (hN : (N : ZMod 3) = 1)
    (he : (e : ZMod 3) = 1) : actualRho 3 N e 3 = 2 := by
  unfold actualRho
  rw [roots_mod_three_e_one hN he]
  norm_num

theorem integerForm_cast (q : ℕ) :
    (integerForm N e p0 q : ZMod l) = actualForm l N e p0 (q : ZMod l) := by
  simp only [integerForm, actualForm, Int.cast_mul, Int.cast_sub, Int.cast_natCast]

theorem integerForm_dvd_iff (q : ℕ) :
    (l : ℤ) ∣ integerForm N e p0 q ↔ actualForm l N e p0 (q : ZMod l) = 0 := by
  rw [← integerForm_cast q, ZMod.intCast_zmod_eq_zero_iff_dvd]

theorem saturation_integerForm_dvd (h : actualRho l N e p0 = l) (q : ℕ) :
    (l : ℤ) ∣ integerForm N e p0 q := by
  exact (integerForm_dvd_iff q).mpr (saturation_forces_actualForm_zero h (q : ZMod l))

theorem saturation_roughCell_empty (J P : Finset ℕ) (hl : l ∈ P)
    (h : actualRho l N e p0 = l) : fourFormRoughCell J P N e p0 = ∅ := by
  apply Finset.eq_empty_iff_forall_not_mem.mpr
  intro q hq
  have hav := (Finset.mem_filter.mp hq).2
  exact hav l hl (saturation_integerForm_dvd h q)
