import Mathlib

/-!
Finite Selberg weights on the actual squarefree prime support. The principal
term is derived by finite Boolean-lattice inversion, rather than postulated.
The definitions take a local root density, later instantiated by four forms.
-/

namespace GoldbachRound17.Selberg
open scoped BigOperators
open Finset
noncomputable section

def sign (s : Finset ℕ) : ℝ := ∏ _p ∈ s, (-1 : ℝ)
def gprod (g : ℕ → ℝ) (s : Finset ℕ) : ℝ := ∏ p ∈ s, g p
def hprod (g : ℕ → ℝ) (s : Finset ℕ) : ℝ := ∏ p ∈ s, g p / (1 - g p)
def primeProduct (s : Finset ℕ) : ℕ := ∏ p ∈ s, p
def support (P : Finset ℕ) (z : ℕ) : Finset (Finset ℕ) :=
  P.powerset.filter fun s => primeProduct s ≤ z
def G (P : Finset ℕ) (z : ℕ) (g : ℕ → ℝ) : ℝ :=
  ∑ s ∈ support P z, hprod g s
def y (P : Finset ℕ) (z : ℕ) (g : ℕ → ℝ) (s : Finset ℕ) : ℝ :=
  if s ∈ support P z then sign s * hprod g s / G P z g else 0
def transform (P : Finset ℕ) (f : Finset ℕ → ℝ) (d : Finset ℕ) : ℝ :=
  ∑ r ∈ P.powerset, if d ⊆ r then sign (r \ d) * f r else 0
def weight (P : Finset ℕ) (z : ℕ) (g : ℕ → ℝ) (d : Finset ℕ) : ℝ :=
  transform P (y P z g) d / gprod g d
def upperSum (P : Finset ℕ) (f : Finset ℕ → ℝ) (r : Finset ℕ) : ℝ :=
  ∑ d ∈ P.powerset, if r ⊆ d then f d else 0
def principal (P : Finset ℕ) (g : ℕ → ℝ) (λ : Finset ℕ → ℝ) : ℝ :=
  ∑ d ∈ P.powerset, ∑ t ∈ P.powerset, λ d * λ t * gprod g (d ∪ t)

lemma sign_square (s : Finset ℕ) : sign s * sign s = 1 := by
  rw [sign, ← prod_mul_distrib]
  simp

lemma alternating_powerset (s : Finset ℕ) :
    ∑ t ∈ s.powerset, sign t = if s = ∅ then 1 else 0 := by
  rw [sign, ← prod_one_add]
  by_cases hs : s = ∅
  · simp [hs]
  · simp [hs, Finset.prod_const_zero, Finset.card_ne_zero.mpr hs]

lemma interval_alternating (r t : Finset ℕ) :
    (∑ d ∈ t.powerset, if r ⊆ d then sign (t \ d) else 0) =
      if r = t then 1 else 0 := by
  classical
  by_cases hrt : r ⊆ t
  · have heq :
        (∑ d ∈ t.powerset, if r ⊆ d then sign (t \ d) else 0) =
          ∑ s ∈ (t \ r).powerset, sign s := by
      rw [← sum_filter]
      apply sum_bij' (fun d _ => t \ d) (fun s _ => t \ s)
      · intro d hd
        simp only [mem_filter, mem_powerset] at hd ⊢
        exact sdiff_subset_sdiff hrt hd.2
      · intro s hs
        simp only [mem_powerset] at hs
        simp only [mem_filter, mem_powerset]
        constructor
        · exact sdiff_subset
        · intro x hx
          simp only [mem_sdiff]
          exact ⟨hrt hx, fun hxs => (mem_sdiff.mp (hs hxs)).2 hx⟩
      · intro d hd
        simp only [mem_filter, mem_powerset] at hd
        ext x
        simp only [mem_sdiff]
        have hdsub := hd.1
        constructor
        · rintro ⟨hxt, hn⟩
          by_contra hxd
          exact hn ⟨hxt, hxd⟩
        · intro hxd
          exact ⟨hdsub hxd, fun h => h.2 hxd⟩
      · intro s hs
        simp only [mem_powerset] at hs
        ext x
        simp only [mem_sdiff]
        constructor
        · rintro ⟨hxt, hn⟩
          by_contra hxs
          exact hn ⟨hxt, hxs⟩
        · intro hxs
          exact ⟨(mem_sdiff.mp (hs hxs)).1, fun h => h.2 hxs⟩
      · intro d hd
        rfl
    rw [heq, alternating_powerset]
    congr 1
    exact propext ⟨fun h => (sdiff_eq_empty_iff_subset.mp h).antisymm hrt,
      fun h => by simp [h]⟩
  · have hne : r ≠ t := fun h => hrt (h ▸ Subset.rfl)
    rw [if_neg hne]
    apply sum_eq_zero
    intro d hd
    exact if_neg fun hrd => hrt (hrd.trans (mem_powerset.mp hd))

lemma inverse_transform (P : Finset ℕ) (f : Finset ℕ → ℝ)
    (r : Finset ℕ) (hr : r ⊆ P) : upperSum P (transform P f) r = f r := by
  classical
  unfold upperSum transform
  simp_rw [ite_sum, ← ite_mul, zero_mul]
  rw [sum_comm]
  have hinner (t : Finset ℕ) (ht : t ∈ P.powerset) :
      (∑ d ∈ P.powerset,
        if r ⊆ d then (if d ⊆ t then sign (t \ d) * f t else 0) else 0) =
        (if r = t then 1 else 0) * f t := by
    have hs :
        (∑ d ∈ P.powerset,
          if r ⊆ d then (if d ⊆ t then sign (t \ d) * f t else 0) else 0) =
          (∑ d ∈ t.powerset, if r ⊆ d then sign (t \ d) else 0) * f t := by
      rw [sum_mul]
      apply sum_subset (powerset_mono.mpr (mem_powerset.mp ht))
      · intro d hd hdt
        have hnot : ¬ d ⊆ t := fun h => hdt (mem_powerset.mpr h)
        simp [hnot]
      · intro d hd
        simp [mem_powerset.mp hd, ite_mul]
    rw [hs, interval_alternating]
  simp_rw [hinner]
  simp [mem_powerset.mpr hr]

lemma gprod_pos {P : Finset ℕ} {g : ℕ → ℝ}
    (hg : ∀ p ∈ P, 0 < g p) {s : Finset ℕ} (hs : s ⊆ P) : 0 < gprod g s := by
  exact prod_pos fun p hp => hg p (hs hp)

lemma hprod_pos {P : Finset ℕ} {g : ℕ → ℝ}
    (hg : ∀ p ∈ P, 0 < g p ∧ g p < 1) {s : Finset ℕ} (hs : s ⊆ P) :
    0 < hprod g s := by
  apply prod_pos
  intro p hp
  exact div_pos (hg p (hs hp)).1 (sub_pos.mpr (hg p (hs hp)).2)

lemma G_pos {P : Finset ℕ} {z : ℕ} {g : ℕ → ℝ} (hz : 1 ≤ z)
    (hg : ∀ p ∈ P, 0 < g p ∧ g p < 1) : 0 < G P z g := by
  unfold G
  apply sum_pos'
  · intro s hs
    exact le_of_lt (hprod_pos hg (mem_powerset.mp (mem_filter.mp hs).1))
  · refine ⟨∅, ?_, ?_⟩
    · simp [support, primeProduct, hz]
    · simp [hprod]

lemma weight_transform {P : Finset ℕ} {z : ℕ} {g : ℕ → ℝ}
    (hg : ∀ p ∈ P, 0 < g p ∧ g p < 1) {r : Finset ℕ} (hr : r ⊆ P) :
    upperSum P (fun d => weight P z g d * gprod g d) r = y P z g r := by
  have heq : ∀ d ∈ P.powerset,
      weight P z g d * gprod g d = transform P (y P z g) d := by
    intro d hd
    unfold weight
    exact div_mul_cancel₀ _ (ne_of_gt (gprod_pos (fun p hp => (hg p hp).1)
      (mem_powerset.mp hd)))
  rw [upperSum, sum_congr rfl (fun d hd => by rw [heq d hd])]
  exact inverse_transform P (y P z g) r hr

end
end GoldbachRound17.Selberg
