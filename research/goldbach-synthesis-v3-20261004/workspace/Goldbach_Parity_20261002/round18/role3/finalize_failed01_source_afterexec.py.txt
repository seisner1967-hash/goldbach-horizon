def sha(p):

def read_json(p):
    return json.loads(Path(p).read_text(encoding='utf-8'))

def write_new_json(p, obj):
    with Path(p).open('x', encoding='utf-8', newline='\n') as f:
        f.write(json.dumps(obj, ensure_ascii=False, indent=2) + '\n')
